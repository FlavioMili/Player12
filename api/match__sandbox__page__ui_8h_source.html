99struct ActivityLayout
100{
101 bool columns;
102 float details, catalogue, page;
103 ActivityLayout(float available, float scale)
104 {
105 const float usable = std::max(scale, available - (6 * scale));
106 columns = usable >= 1072 * scale;
107 details = columns ? (usable >= 1152 * scale ? 886 : 806) * scale : usable;
108 if (!columns)
109 {
110 for (float width : {886.0F, 806.0F, 660.0F})
111 {
112 if (usable >= width * scale)
113 {
114 details = width * scale;
115 break;
116 }
117 }
118 }
119 catalogue = columns ? 260 * scale : details;
120 page = columns ? catalogue + (6 * scale) + details : details;
121 }
122};

309inline bool choice(const char* label, bool selected, float scale, ImFont* bold)
310{
311 const ImVec2 a = ImGui::GetCursorScreenPos();
312 const float width = ImGui::GetContentRegionAvail().x;
313 const bool clicked = ImGui::InvisibleButton(label, ImVec2(width, 36 * scale),
314 ImGuiButtonFlags_EnableNav);
315 auto* draw = ImGui::GetWindowDrawList();
316 if (selected || ImGui::IsItemHovered())
317 {
318 draw->AddRectFilled(a, ImVec2(a.x + width, a.y + (36 * scale)),
319 selected ? IM_COL32(46, 39, 32, 255) : MID);
320 }
321 if (selected)
322 {
323 draw->AddRectFilled(a, ImVec2(a.x + (3 * scale), a.y + (36 * scale)),
324 CREAM);
325 }
326 if (ImGui::IsItemFocused())
327 {
328 draw->AddRect(a, ImVec2(a.x + width, a.y + (36 * scale)), LIME);
329 }
330 draw->PushClipRect(a, ImVec2(a.x + width, a.y + (36 * scale)), true);
331 SandboxTypography::label(draw, selected ? bold : ImGui::GetFont(), 16, label,
332 a, ImVec2(a.x + width, a.y + (36 * scale)), CREAM, 0,
333 false, 12 * scale);
334 draw->PopClipRect();
