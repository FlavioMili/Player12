122inline void fillMatchRenderSnapshot(const MatchEngine& engine,
123 MatchRenderSnapshot& snapshot)
124{
125 const auto& players = engine.getPlayers();
126 const auto& previousPositions = engine.getPreviousPlayerPositions();
127 const auto& previousFacingAngles = engine.getPreviousPlayerFacingAngles();
128 snapshot.players.clear();
129 snapshot.players.reserve(players.size());
130 for (std::size_t index = 0; index < players.size(); ++index)
131 {
132 const MatchPlayer& source = players[index];
133 MatchRenderPlayer& renderPlayer = snapshot.players.emplace_back();
134 renderPlayer.player = source.player;
135 renderPlayer.isHomeTeam = source.isHomeTeam;
136 renderPlayer.currentPosition = source.position;
137 renderPlayer.movementTarget = source.movementTarget;
138 renderPlayer.intent = source.intent;
139 renderPlayer.currentFacingAngle = source.facingAngle;
140 renderPlayer.stamina = source.stamina;
141 renderPlayer.possessesBall = source.player != nullptr &&
142 engine.getBall().possessedBy == source.player;
143 renderPlayer.onPitch = source.onPitch;
144 renderPlayer.isGoalkeeper = source.isGoalkeeper;
145 renderPlayer.isDiving = source.isDiving;
146 renderPlayer.isInjured = source.isInjured;
147 renderPlayer.yellowCards = source.yellowCards;
148 renderPlayer.heightMetres = source.heightMetres;
149 renderPlayer.speedMetresPerSecond =
150 std::sqrt(source.velocity.x * source.velocity.x +
151 source.velocity.y * source.velocity.y);
152 renderPlayer.tackleCooldown = source.tackleCooldown;
153 renderPlayer.previousPosition = index < previousPositions.size()
154 ? previousPositions[index]
155 : renderPlayer.currentPosition;
156 renderPlayer.previousFacingAngle = index < previousFacingAngles.size()
157 ? previousFacingAngles[index]
158 : renderPlayer.currentFacingAngle;
159 }
160
161 const MatchBall& ball = engine.getBall();
162 snapshot.ball.currentPosition = ball.position;
163 snapshot.ball.previousPosition = engine.getPreviousBallPosition();
164 snapshot.ball.currentZ = ball.z;
165 snapshot.ball.previousZ = engine.getPreviousBallZ();
166 snapshot.ball.possessedBy = ball.possessedBy;
167 snapshot.ball.currentHeightMetres =
168 ball.z * MatchTuning::Units::BALL_Z_METRES;
169 snapshot.ball.previousHeightMetres =
170 engine.getPreviousBallZ() * MatchTuning::Units::BALL_Z_METRES;
171 snapshot.ball.isShot = ball.isShot;
172 snapshot.ball.isAerialDelivery = ball.isAerialDelivery;
173 snapshot.ball.fromThrowIn = ball.fromThrowIn;
174 snapshot.ball.kicker = ball.kicker;
175 snapshot.ball.kickerLockout = ball.kickerLockout;
176
177 snapshot.state = engine.getState();
178 snapshot.homePhase = engine.getHomePhase();
179 snapshot.awayPhase = engine.getAwayPhase();
180 snapshot.transitionSecondsRemaining = engine.getTransitionSecondsRemaining();
181 snapshot.homeScore = engine.getHomeScore();
182 snapshot.awayScore = engine.getAwayScore();
183 snapshot.goalScoredByHome = engine.getGoalScoredByHome();
184 snapshot.goalCelebrationRemaining = engine.getGoalCelebrationRemaining();
185 snapshot.goalCelebrationDuration = engine.getGoalCelebrationDuration();
186 snapshot.matchTimeMinutes = engine.getMatchTimeMinutes();
187 snapshot.events = &engine.getEvents();
188 snapshot.stats = &engine.getStats();
189 snapshot.interpolationAlpha = engine.getInterpolationAlpha();
190 snapshot.period = engine.getPeriod();
191 snapshot.addedMinutesFirstHalf = engine.getAddedMinutes(1);
192 snapshot.addedMinutesSecondHalf = engine.getAddedMinutes(2);
193 snapshot.homeGoalkeeperState = engine.getHomeGoalkeeperState();
194 snapshot.awayGoalkeeperState = engine.getAwayGoalkeeperState();
195 snapshot.playerStats = &engine.getPlayerStats();
196}
