432template <typename T>
433int compare(const T& left, const T& right)
434{
435 if constexpr (std::is_same_v<T, bool>)
436 {
437 if (left == right) return 0;
438 return left ? 1 : -1;
439 }
440 else
441 {
442 if (left < right) return -1;
443 if (right < left) return 1;
444 return 0;
445 }
446}
