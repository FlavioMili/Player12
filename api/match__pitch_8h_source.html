41struct PitchGoal
42{
43 GoalKind kind = GoalKind::FULL;
44 [[nodiscard]] constexpr bool present() const
45 {
46 return kind != GoalKind::NONE;
47 }
49 float halfWidth = MatchTuning::Shooting::GOAL_HALF_WIDTH_METRES;
51 float crossbar = MatchTuning::Units::CROSSBAR_HEIGHT_METRES;
53 float post = MatchTuning::Pitch::GOAL_POST_METRES;
55 float netDepth = MatchTuning::Pitch::GOAL_NET_DEPTH_METRES;
56
62 [[nodiscard]] static constexpr PitchGoal ofKind(GoalKind kind)
63 {
64 const auto sized = [kind](float width, float height)
65 {
66 constexpr float FULL_HEIGHT = MatchTuning::Units::CROSSBAR_HEIGHT_METRES;
67 PitchGoal goal;
68 goal.kind = kind;
69 goal.halfWidth = 0.5F * width;
70 goal.crossbar = height;
71 goal.post = MatchTuning::Pitch::GOAL_POST_METRES * height / FULL_HEIGHT;
72 goal.netDepth =
73 MatchTuning::Pitch::GOAL_NET_DEPTH_METRES * height / FULL_HEIGHT;
74 return goal;
75 };
76 switch (kind)
77 {
78 case GoalKind::NONE:
79 {
80 PitchGoal none;
81 none.kind = GoalKind::NONE;
82 return none;
83 }
84 case GoalKind::FULL:
85 return PitchGoal{};
86 case GoalKind::YOUTH_9V9:
87 return sized(4.88F, 2.13F);
88 case GoalKind::YOUTH_7V7:
89 return sized(3.66F, 1.83F);
90 case GoalKind::FUTSAL:
91 return sized(3.0F, 2.0F);
92 case GoalKind::FIVE_A_SIDE:
93 return sized(3.66F, 1.22F);
94 case GoalKind::SMALL_FRAMED:
95 return sized(1.8F, 1.2F);
96 case GoalKind::POP_UP:
97 {
98 PitchGoal goal = sized(1.2F, 0.8F);
99 goal.post = 0.03F;
100 goal.netDepth = 0.6F;
101 return goal;
102 }
103 }
104 return PitchGoal{};
105 }
106};

374struct PitchPlacement
375{
376 PitchPoint centre{};

382 static constexpr float QUARTER_TURN = 1.5707963F;

384 [[nodiscard]] static constexpr PitchPlacement of(const MatchPitch& pitch,
385 PitchPoint centre,
386 bool turned)
387 {
388 return PitchPlacement{.centre = centre,
389 .turned = turned,
390 .lengthMetres = pitch.length,
391 .widthMetres = pitch.width};
392 }
394 [[nodiscard]] constexpr PitchPoint toGround(PitchPoint local) const
395 {
396 const float dx = local.x - (0.5F * lengthMetres);
397 const float dy = local.y - (0.5F * widthMetres);
398 return turned ? PitchPoint{centre.x - dy, centre.y + dx}
399 : PitchPoint{centre.x + dx, centre.y + dy};
400 }
401 [[nodiscard]] constexpr PitchPoint toLocal(PitchPoint ground) const
402 {
403 const float gx = ground.x - centre.x;
404 const float gy = ground.y - centre.y;
405 return turned ? PitchPoint{(0.5F * lengthMetres) + gy,
406 (0.5F * widthMetres) - gx}
407 : PitchPoint{(0.5F * lengthMetres) + gx,
408 (0.5F * widthMetres) + gy};
409 }
412 [[nodiscard]] constexpr Vector2F toGround(Vector2F local) const
413 {
414 return turned ? Vector2F{.x = -local.y, .y = local.x} : local;
415 }
416 [[nodiscard]] constexpr Vector2F toLocal(Vector2F ground) const
417 {
418 return turned ? Vector2F{.x = ground.y, .y = -ground.x} : ground;
419 }
423 [[nodiscard]] constexpr float toGroundAngle(float angle) const
424 {
425 return turned ? angle + QUARTER_TURN : angle;
426 }
427 [[nodiscard]] constexpr float toLocalAngle(float angle) const
428 {
429 return turned ? angle - QUARTER_TURN : angle;
430 }
431};
