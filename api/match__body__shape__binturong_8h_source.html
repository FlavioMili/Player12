90inline LegPose legPoseOf(float heightMetres, Vector2F velocity, Vector2F facing,
91 const std::array<float, 6>& rolls)
92{
93 using S = MatchTuning::Stride;
94 const float scale = heightMetres / S::REFERENCE_HEIGHT_METRES;
95 const float hipHeight =
96 MatchTuning::Ball::CHEST_LINE_HEIGHT_SHARE * heightMetres;
97 const float kneeHeight =
98 MatchTuning::Ball::FEET_LINE_HEIGHT_SHARE * heightMetres;
99 const float ankleHeight = S::FOOT_HEIGHT * scale;
100 const auto symmetric = [](float roll) { return (2.0F * roll) - 1.0F; };
101
102 const float speed = std::hypot(velocity.x, velocity.y);
103 const Vector2F travel =
104 speed > 1e-4F ? Vector2F{.x = velocity.x / speed, .y = velocity.y / speed}
105 : facing;
106 const float offFacing =
107 std::acos(std::clamp((travel.x * facing.x) + (travel.y * facing.y), -1.0F,
108 1.0F)) *
109 57.2957795F;
110
111 enum class Kind : std::uint8_t
112 {
113 STANDING,
114 FORWARD,
115 SIDESTEP,
116 BACKPEDAL
117 };
118 Kind kind = Kind::FORWARD;
119 int gait = 0; // walk, jog, run, sprint (a backpedal as a walk)
120 float step = S::STEP_WALK;
121 if (speed < S::STANDING_SPEED)
122 {
123 kind = Kind::STANDING;
124 step = S::STANDING_FEET_APART;
125 }
126 else if (speed <= S::SHUFFLE_TOP_SPEED && offFacing > S::BACKPEDAL_DEGREES)
127 {
128 kind = Kind::BACKPEDAL;
129 step = S::STEP_BACKPEDAL;
130 }
131 else if (speed <= S::SHUFFLE_TOP_SPEED && offFacing > S::SIDESTEP_DEGREES)
132 {
133 kind = Kind::SIDESTEP;
134 step = S::STEP_SIDESTEP;
135 }
136 else
137 {
138 // Forward, or on a diagonal: his legs spread along his travel.
139 constexpr float STEPS[4] = {S::STEP_WALK, S::STEP_JOG, S::STEP_RUN,
140 S::STEP_SPRINT};
141 gait = speed <= S::WALK_TOP_SPEED ? 0
142 : speed <= S::JOG_TOP_SPEED ? 1
143 : speed <= S::RUN_TOP_SPEED ? 2
144 : 3;
145 step = STEPS[gait];
146 }
147
148 // Hips side by side across his body; the legs spread along an axis
149 // (standing: across him too), turned a little by the noise.
150 const Vector2F across{.x = -facing.y, .y = facing.x};
151 Vector2F axis = kind == Kind::STANDING ? across : travel;
152 const float turn = symmetric(rolls[0]) * S::NOISE_DEGREES * 0.0174532925F;
153 axis = {.x = (axis.x * std::cos(turn)) - (axis.y * std::sin(turn)),
154 .y = (axis.x * std::sin(turn)) + (axis.y * std::cos(turn))};
155 const Vector2F sideways{.x = -axis.y, .y = axis.x};
156 const float spread =
157 step * scale * (1.0F + symmetric(rolls[3]) * S::NOISE_SPREAD);
158
159 // Each leg: where its foot is along the axis and how far it is lifted;
160 // a passing swing leg folds (knee ahead, heel up behind).
161 struct Foot
162 {
163 float along = 0.0F;
164 float rise = 0.0F;
165 bool folded = false;
166 };
167 std::array<Foot, 2> feet{};
168 const std::size_t lead = rolls[1] < 0.5F ? 0 : 1;
169 const std::size_t other = 1 - lead;
170 const float point = rolls[2];
171 if (kind == Kind::STANDING)
172 {
173 feet[0].along = 0.5F * spread;
174 feet[1].along = -0.5F * spread;
175 }
176 else if (kind == Kind::SIDESTEP)
177 {
178 const float apart =
179 point < S::SIDESTEP_SHARE_WIDE ? spread
180 : point < S::SIDESTEP_SHARE_WIDE + S::SIDESTEP_SHARE_HALF
181 ? 0.5F * spread
182 : S::SIDESTEP_TOGETHER * scale;
183 feet[0].along = 0.5F * apart;
184 feet[1].along = -0.5F * apart;
185 }
186 else
187 {
188 const float full = S::SHARE_FULL[gait];
189 const float part = S::SHARE_PART[gait];
190 if (point < full)
191 {
192 feet[lead].along = 0.5F * spread;
193 feet[other].along = -0.5F * spread;
194 }
195 else if (point < full + part)
196 {
197 // Opening (the swing foot reaching ahead) or closing (just off the
198 // ground behind): the planted foot opposite.
199 const bool opening = point < full + 0.5F * part;
200 const float half = 0.5F * S::PART_SPREAD_SHARE * spread;
201 feet[lead].along = opening ? half : -half;
202 feet[lead].rise = S::SWING_RISE_PART[gait];
203 feet[other].along = opening ? -half : half;
204 }
205 else
206 {
207 feet[lead].along = -S::PASSING_ANKLE_BEHIND * scale;
208 feet[lead].rise = S::SWING_RISE_PASSING[gait];
209 feet[lead].folded = true;
210 }
211 }
212
213 LegPose pose;
214 const Vector2F footFacing = kind == Kind::FORWARD ? axis : facing;
215 const float runningApart = kind == Kind::FORWARD || kind == Kind::BACKPEDAL
216 ? 0.5F * S::RUNNING_FEET_APART * scale
217 : 0.0F;
218 for (std::size_t index = 0; index < 2; ++index)
219 {
220 const float hipSide = (index == 0 ? 0.5F : -0.5F) * S::HIPS_APART * scale;
221 // His feet stay on their own hip's side of the line they run along.
222 const float sideSign =
223 (across.x * sideways.x + across.y * sideways.y) * hipSide >= 0.0F
224 ? 1.0F
225 : -1.0F;
226 const float side = (sideSign * runningApart) +
227 (symmetric(rolls[4 + index]) * S::NOISE_SIDEWAYS);
228 const Foot& foot = feet[index];
229 LegPose::Leg& leg = pose.legs[index];
230 leg.hip = {
231 .x = across.x * hipSide, .y = across.y * hipSide, .z = hipHeight};
232 leg.footFacing = footFacing;
233 leg.ankle = {.x = (axis.x * foot.along) + (sideways.x * side),
234 .y = (axis.y * foot.along) + (sideways.y * side),
235 .z = ankleHeight + foot.rise};
236 if (foot.folded)
237 {
238 const float ahead = S::PASSING_KNEE_AHEAD * scale;
239 leg.knee = {.x = leg.hip.x + (axis.x * ahead),
240 .y = leg.hip.y + (axis.y * ahead),
241 .z = kneeHeight + (0.5F * foot.rise)};
242 }
243 else
244 {
245 leg.knee = {.x = 0.5F * (leg.hip.x + leg.ankle.x),
246 .y = 0.5F * (leg.hip.y + leg.ankle.y),
247 .z = 0.5F * (leg.hip.z + leg.ankle.z)};
248 }
249 }
250 return pose;
251}

255inline Point3F hipOf(float heightMetres, Vector2F facing, std::size_t leg)
256{
257 const float scale =
258 heightMetres / MatchTuning::Stride::REFERENCE_HEIGHT_METRES;
259 const float side =
260 (leg == 0 ? 0.5F : -0.5F) * MatchTuning::Stride::HIPS_APART * scale;
261 return {.x = -facing.y * side,
262 .y = facing.x * side,
263 .z = MatchTuning::Ball::CHEST_LINE_HEIGHT_SHARE * heightMetres};
264}

269inline LegPose blockLegPoseOf(float heightMetres, Vector2F facing,
270 std::size_t leg, Point3F aim, Vector2F footFacing,
271 float outShare)
272{
273 const float ankleHeight = MatchTuning::Stride::FOOT_HEIGHT * heightMetres /
274 MatchTuning::Stride::REFERENCE_HEIGHT_METRES;
275 LegPose pose;
276 for (std::size_t index = 0; index < 2; ++index)
277 {
278 LegPose::Leg& each = pose.legs[index];
279 each.hip = hipOf(heightMetres, facing, index);
280 const Point3F under{.x = each.hip.x, .y = each.hip.y, .z = ankleHeight};
281 each.ankle = under;
282 each.footFacing = facing;
283 if (index == leg)
284 {
285 const float out = std::clamp(outShare, 0.0F, 1.0F);
286 each.ankle = {.x = under.x + ((aim.x - under.x) * out),
287 .y = under.y + ((aim.y - under.y) * out),
288 .z = under.z + ((aim.z - under.z) * out)};
289 each.footFacing = footFacing;
290 }
291 each.knee = {.x = 0.5F * (each.hip.x + each.ankle.x),
292 .y = 0.5F * (each.hip.y + each.ankle.y),
293 .z = 0.5F * (each.hip.z + each.ankle.z)};
294 }
295 return pose;
