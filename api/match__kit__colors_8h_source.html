90inline ImU32 shadeColor(ImU32 color, float factor)
91{
92 const auto scale = [factor](ImU32 value, int shift)
93 {
94 const float scaled =
95 static_cast<float>((value >> shift) & 0xFFU) * factor + 0.5f;
96 float bounded = scaled;
97 if (scaled < 0.0f)
98 bounded = 0.0f;
99 else if (scaled > 255.0f)
100 bounded = 255.0f;
101 return static_cast<ImU32>(bounded) << shift;
102 };
103 return scale(color, IM_COL32_R_SHIFT) | scale(color, IM_COL32_G_SHIFT) |
104 scale(color, IM_COL32_B_SHIFT) | (color & IM_COL32_A_MASK);
105}

108inline ImU32 mixColor(ImU32 first, ImU32 second, float t)
109{
110 const auto blend = [t](ImU32 a, ImU32 b, int shift)
111 {
112 const auto from = static_cast<float>((a >> shift) & 0xFFU);
113 const auto to = static_cast<float>((b >> shift) & 0xFFU);
114 return static_cast<ImU32>(std::lerp(from, to, t) + 0.5f) << shift;
115 };
116 return blend(first, second, IM_COL32_R_SHIFT) |
117 blend(first, second, IM_COL32_G_SHIFT) |
118 blend(first, second, IM_COL32_B_SHIFT) |
119 blend(first, second, IM_COL32_A_SHIFT);
120}

126constexpr ImU32 mixColor256(ImU32 first, ImU32 second, std::uint32_t t256)
127{
128 const std::uint32_t keep = 256U - t256;
129 const std::uint32_t evenChannels =
130 (((first & 0x00FF00FFU) * keep + (second & 0x00FF00FFU) * t256) >> 8U) &
131 0x00FF00FFU;
132 const std::uint32_t oddChannels = ((((first >> 8U) & 0x00FF00FFU) * keep +
133 ((second >> 8U) & 0x00FF00FFU) * t256) >>
134 8U) &
135 0x00FF00FFU;
136 return evenChannels | (oddChannels << 8U);
137}

139constexpr ImU32 shadeColor256(ImU32 color, std::uint32_t factor256)
140{
141 const auto channel = [&](int shift)
142 {
143 const std::uint32_t value = (((color >> shift) & 0xFFU) * factor256) >> 8U;
144 return (value > 255U ? 255U : value) << shift;
145 };
146 return channel(IM_COL32_R_SHIFT) | channel(IM_COL32_G_SHIFT) |
147 channel(IM_COL32_B_SHIFT) | (color & IM_COL32_A_MASK);
148}
