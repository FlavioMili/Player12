133[[nodiscard]] constexpr float dugoutRoofHeight(const Rect& dugout, float y)
134{
135 using P = MatchTuning::Pitch;
136 const float share = (y - dugout.y0) / (dugout.y1 - dugout.y0);
137 const float clamped = share < 0.0F ? 0.0F : share > 1.0F ? 1.0F : share;
138 return P::DUGOUT_ROOF_FRONT_METRES +
139 ((P::DUGOUT_ROOF_BACK_METRES - P::DUGOUT_ROOF_FRONT_METRES) * clamped);
140}
