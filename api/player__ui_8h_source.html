36inline void detailPanel(const char* id, const Player* player,
37 const StatsConfig& statsConfig,
38 const Player* comparison = nullptr, float height = 0.0f)
39{
40 const Theme::Palette& palette = Theme::palette();
41 const float dpi = ImGui::GetStyle().FontScaleDpi;
42 // A negative height sizes the card to its content (inside a scrolling
43 // parent), 0 fills the remaining space.
44 if (height < 0.0f)
45 UI::beginAutoHeightCard(id, nullptr);
46 else
47 UI::beginCard(id, nullptr, ImVec2(0.0f, height), true);
48 if (!player)
49 {
50 ImGui::PushStyleColor(ImGuiCol_Text, palette.muted);
51 ImGui::TextWrapped("%s", LOC("PLAYER_SELECT_PROMPT"));
52 ImGui::PopStyleColor();
53 UI::endCard();
54 return;
55 }
56
57 const double overall = player->getOverall(statsConfig);
58 UI::ratingChip(overall);
59 ImGui::SameLine();
60 {
61 Theme::ScopedText title(Theme::Text::TITLE);
62 ImGui::TextUnformatted(player->getName().c_str());
63 }
64 ImGui::PushStyleColor(ImGuiCol_Text, palette.muted);
65 ImGui::TextWrapped(
66 "%s · %s %d · %d cm · %s", RoleUtils::longName(player->getRole()),
67 LOC("PLAYER_AGE"), player->getAge(), player->getHeight(),
68 player->getFoot() == Foot::Right ? LOC("PLAYER_FOOT_RIGHT")
69 : LOC("PLAYER_FOOT_LEFT"));
70 ImGui::PopStyleColor();
71
72 const float keyWidth = 130.0f * dpi;
73 const std::string wage = Format::money(player->getWage());
74 UI::keyValue(LOC("PLAYER_WEEKLY_WAGE"), wage.c_str(), keyWidth);
75 const std::string contract = std::to_string(player->getContractYears());
76 ImGui::PushStyleColor(ImGuiCol_Text, player->getContractYears() <= 1
77 ? palette.negative
78 : palette.text);
79 UI::keyValue(LOC("PLAYER_CONTRACT"), contract.c_str(), keyWidth);
80 ImGui::PopStyleColor();
81
82 if (comparison)
83 {
84 const double delta = overall - comparison->getOverall(statsConfig);
85 ImGui::TextColored(delta >= 0.0 ? palette.positive : palette.negative,
86 "%s: %+.1f", LOC("PLAYER_OVERALL_CHANGE"), delta);
87 if (player->getRole() == comparison->getRole())
88 {
89 ImGui::TextColored(palette.positive, "%s", LOC("PLAYER_ROLE_MATCH"));
90 }
91 else
92 {
93 ImGui::TextColored(palette.warning, "%s: %s -> %s",
94 LOC("PLAYER_ROLE_CHANGE"),
95 RoleUtils::shortName(comparison->getRole()),
96 RoleUtils::shortName(player->getRole()));
97 }
98 }
99
100 ImGui::Dummy(ImVec2(0.0f, Theme::Space::XS * dpi));
101 UI::sectionLabel(LOC("PLAYER_ATTRIBUTES"));
102 const float labelWidth = 100.0f * dpi;
103 for (const auto& [name, value] : player->getStats())
104 {
105 const std::string label = PlayerView::statLabel(name);
106 if (!comparison)
107 {
108 UI::attributeBar(label.c_str(), value, labelWidth);
109 continue;
110 }
111 const auto other = comparison->getStats().find(name);
112 const float delta =
113 value - (other != comparison->getStats().end() ? other->second : 0.0f);
114 const std::string deltaText = std::format("{:+.0f}", delta);
115 const float deltaWidth = ImGui::CalcTextSize("+00").x;
116 const float rowStart = ImGui::GetCursorPosX();
117 const float available = ImGui::GetContentRegionAvail().x;
118 UI::attributeBar(label.c_str(), value, labelWidth, 100.0f,
119 available - deltaWidth - Theme::Space::S * dpi);
120 ImGui::SameLine(rowStart + available - deltaWidth);
121 ImGui::TextColored(delta >= 0.0f ? palette.positive : palette.negative,
122 "%s", deltaText.c_str());
123 }
124 UI::endCard();
125}
