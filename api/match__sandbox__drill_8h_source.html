40inline const char* drillCategoryName(DrillCategory category)
41{
42 switch (category)
43 {
44 case DrillCategory::Movement:
45 return "MOVEMENT";
46 case DrillCategory::Passing:
47 return "PASSING";
48 case DrillCategory::Shooting:
49 return "SHOOTING";
50 case DrillCategory::Control:
51 return "CONTROL";
52 case DrillCategory::Duels:
53 return "DUELS";
54 case DrillCategory::Aerial:
55 return "AERIAL";
56 case DrillCategory::Goalkeeping:
57 return "GOALKEEPING";
58 case DrillCategory::Count:
59 break;
60 }
61 return "DRILLS";
62}
