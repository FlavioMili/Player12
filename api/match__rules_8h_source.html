152inline float staminaDrainPerSecond(float speedRatio, float endurance,
153 float pressingIntensity)
154{
155 using P = MatchTuning::Fatigue;
156 const float ratio = std::clamp(speedRatio, 0.0f, 1.5f);
157 const float sprint = std::max(0.0f, ratio - P::SPRINT_THRESHOLD) /
158 (1.0f - P::SPRINT_THRESHOLD);
159 const float work =
160 P::IDLE_DRAIN + ratio * ratio * P::RUNNING_DRAIN +
161 sprint * P::SPRINT_DRAIN +
162 std::clamp(pressingIntensity, 0.0f, 1.0f) * P::PRESSING_DRAIN;
163 return work * (P::ENDURANCE_BASE -
164 std::clamp(endurance, 0.0f, 1.0f) * P::ENDURANCE_RELIEF);
165}
