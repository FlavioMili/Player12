173inline float staminaDrainPerSecond(float speedRatio, float endurance,
174 float pressingIntensity)
175{
176 using P = MatchTuning::Fatigue;
177 const float ratio = std::clamp(speedRatio, 0.0F, 1.5F);
178 const float sprint = std::max(0.0F, ratio - P::SPRINT_THRESHOLD) /
179 (1.0F - P::SPRINT_THRESHOLD);
180 const float work =
181 P::IDLE_DRAIN + (ratio * ratio * P::RUNNING_DRAIN) +
182 (sprint * P::SPRINT_DRAIN) +
183 (std::clamp(pressingIntensity, 0.0F, 1.0F) * P::PRESSING_DRAIN);
184 return work * (P::ENDURANCE_BASE -
185 std::clamp(endurance, 0.0F, 1.0F) * P::ENDURANCE_RELIEF);
186}

194inline float walkRunTransitionSpeed(float heightMetres)
195{
196 constexpr float GRAVITY = 9.81F;
197 constexpr float TRANSITION_FROUDE = 0.5F;
198 constexpr float LEG_LENGTH_SHARE = 0.53F;
199 return std::sqrt(TRANSITION_FROUDE * GRAVITY * LEG_LENGTH_SHARE *
200 std::max(heightMetres, 1.0F));
201}
