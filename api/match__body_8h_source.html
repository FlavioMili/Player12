97constexpr std::optional<std::array<BodyId, 2>> parseBodyPair(
98 std::string_view text)
99{
100 const std::size_t comma = text.find(',');
101 const auto home = bodyFromKey(text.substr(0, comma));
102 if (!home)
103 {
104 return std::nullopt;

106 if (comma == std::string_view::npos)

108 return std::array{*home, *home};
109 }
110 const auto away = bodyFromKey(text.substr(comma + 1));
111 if (!away)
112 {
113 return std::nullopt;
114 }
115 return std::array{*home, *away};
