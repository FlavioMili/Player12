28// Button/navigation labels are ASCII. Use visible glyph bounds, including side
29// bearings, rather than the font's line box or its trailing advance. Account
30// for the renderer snapping each separately tracked character to a pixel.
31inline LabelBounds labelBounds(ImFont* font, float pixelSize, const char* label,
32 float tracking, bool snap = true)
33{
34 LabelBounds bounds;
35 auto* baked = font->GetFontBaked(pixelSize);
36 const float factor = pixelSize / baked->Size;
37 float x = 0;
38 for (const unsigned char* c = reinterpret_cast<const unsigned char*>(label);
39 *c != 0u; ++c)
40 {
41 const auto* glyph = baked->FindGlyph(*c);
42 const float gx = snap ? std::floor(x) : x;
43 if (glyph->Visible)
44 {
45 bounds.min.x = std::min(bounds.min.x, gx + (glyph->X0 * factor));
46 bounds.min.y = std::min(bounds.min.y, glyph->Y0 * factor);
47 bounds.max.x = std::max(bounds.max.x, gx + (glyph->X1 * factor));
48 bounds.max.y = std::max(bounds.max.y, glyph->Y1 * factor);
49 bounds.visible = true;
50 }
51 x += glyph->AdvanceX * factor + tracking;
52 }
53 if (!bounds.visible)
54 {
55 bounds.min = bounds.max = ImVec2(0, 0);
56 }
57 return bounds;
58}

60inline void label(ImDrawList* draw, ImFont* font, float size, const char* text,
61 ImVec2 a, ImVec2 b, ImU32 color, float tracking = 0,
62 bool centerHorizontal = true, float leftInset = 0)
63{
64 ImGui::PushFont(font, size);
65 const float pixels = ImGui::GetFontSize();
66 const bool snap = (draw->Flags & ImDrawListFlags_TextNoPixelSnap) == 0;
67 const auto bounds = labelBounds(font, pixels, text, tracking, snap);
68 ImVec2 origin(centerHorizontal ? (a.x + b.x - bounds.min.x - bounds.max.x) / 2
69 : a.x + leftInset - bounds.min.x,
70 (a.y + b.y - bounds.min.y - bounds.max.y) / 2);
71 if (snap)
72 {
73 origin.x = std::round(origin.x);
74 origin.y = std::round(origin.y);
75 }
76 auto* baked = font->GetFontBaked(pixels);
77 const float factor = pixels / baked->Size;
78 float x = 0;
79 for (const char* c = text; *c != 0; ++c)
80 {
81 const auto* glyph = baked->FindGlyph(static_cast<unsigned char>(*c));
82 const float advance = glyph->AdvanceX * factor;
83 draw->AddText(font, pixels, ImVec2(origin.x + x, origin.y), color, c,
84 c + 1);
85 x += advance + tracking;
86 }
87 ImGui::PopFont();
88}
