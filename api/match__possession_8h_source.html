143constexpr PossessionSnapshot interpretLegacyPossession(PossessionSnapshot state)
144{
145 state.controller = 0;
146 state.claimant = state.inTouch ? state.legacyOwner : 0;
147 state.mode = PossessionMode::UNCONTROLLED;
148 if (state.legacyOwner == 0)
149 {
150 return state;
151 }
152 if (!state.openPlay)
153 {
154 state.mode = PossessionMode::RESTART_OWNER;
155 }
156 else if (state.inHands)
157 {
158 state.mode = PossessionMode::KEEPER_HANDS;
159 state.controller = state.legacyOwner;
160 }
161 else if (state.inTouch)
162 {
163 const bool controlled = state.footControlled.value_or(state.collected);
164 state.mode =
165 controlled ? PossessionMode::FOOT_CONTROL : PossessionMode::TOUCH_CLAIM;
166 if (controlled)
167 {
168 state.controller = state.legacyOwner;
169 }
170 }
171 else
172 {
173 state.mode = PossessionMode::LEGACY_ATTACHED;
174 }
175 return state;
176}

181enum class PossessionIssue : std::uint32_t
182{
183 TOUCH_WITHOUT_OWNER = 1U << 0,
184 TOUCH_OUTSIDE_OPEN_PLAY = 1U << 1,
185 COLLECTION_WITHOUT_TOUCH = 1U << 2,
186 DROPPING_WITHOUT_TOUCH = 1U << 3,
187 HANDS_WITHOUT_OWNER = 1U << 4,
188 HANDS_AND_TOUCH = 1U << 5,
189 OWNER_UNKNOWN = 1U << 6,
190 OWNER_INACTIVE = 1U << 7,
191 HANDS_NOT_KEEPER = 1U << 8,
192 HANDS_OUTSIDE_AREA = 1U << 9,
193 MEET_OWNER_MISMATCH = 1U << 10,
194 PASS_AND_SHOT = 1U << 11,
195};
196
197constexpr std::uint32_t possessionIssues(const PossessionSnapshot& state)
198{
199 std::uint32_t issues = 0;
200 const auto add = [&](bool failed, PossessionIssue issue)
201 {
202 if (failed)
203 {
204 issues |= static_cast<std::uint32_t>(issue);
205 }
206 };
207 add(state.inTouch && state.legacyOwner == 0,
208 PossessionIssue::TOUCH_WITHOUT_OWNER);
209 add(state.inTouch && !state.openPlay,
210 PossessionIssue::TOUCH_OUTSIDE_OPEN_PLAY);
211 add(state.collected && !state.inTouch,
212 PossessionIssue::COLLECTION_WITHOUT_TOUCH);
213 add(state.dropping && !state.inTouch,
214 PossessionIssue::DROPPING_WITHOUT_TOUCH);
215 add(state.inHands && state.legacyOwner == 0,
216 PossessionIssue::HANDS_WITHOUT_OWNER);
217 add(state.inHands && state.inTouch, PossessionIssue::HANDS_AND_TOUCH);
218 add(state.legacyOwner != 0 && !state.ownerKnown,
219 PossessionIssue::OWNER_UNKNOWN);
220 add(state.legacyOwner != 0 && state.ownerKnown && !state.ownerActive,
221 PossessionIssue::OWNER_INACTIVE);
222 add(state.inHands && state.ownerKnown && !state.ownerGoalkeeper,
223 PossessionIssue::HANDS_NOT_KEEPER);
224 add(state.inHands && state.ownerKnown && !state.ownerInOwnArea,
225 PossessionIssue::HANDS_OUTSIDE_AREA);
226 add(state.meetFor != 0 &&
227 (!state.inTouch || state.meetFor != state.legacyOwner),
228 PossessionIssue::MEET_OWNER_MISMATCH);
229 add(state.pass && state.shot, PossessionIssue::PASS_AND_SHOT);
230 return issues;
231}
