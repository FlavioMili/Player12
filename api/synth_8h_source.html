30class Noise
31{
32 public:
33 explicit Noise(std::uint32_t seed = 0x9E3779B9U) : state(seed | 1U) {}
34 float next()
35 {
36 state ^= state << 13U;
37 state ^= state >> 17U;
38 state ^= state << 5U;
39 return static_cast<float>(static_cast<std::int32_t>(state)) *
40 (1.0f / 2147483648.0f);
41 }
43 float unit() { return 0.5f * (next() + 1.0f); }
44 void reseed(std::uint32_t seed) { state = seed | 1U; }
45
46 private:
47 std::uint32_t state;
48};

51class Svf
52{
53 public:
54 struct Output
55 {
56 float low = 0.0f;
57 float band = 0.0f;
58 float high = 0.0f;
59 };
60
61 void set(float cutoffHz, float q);
62 Output process(float input)
63 {
64 const float v3 = input - ic2;
65 const float v1 = a1 * ic1 + a2 * v3;
66 const float v2 = ic2 + a2 * ic1 + a3 * v3;
67 ic1 = 2.0f * v1 - ic1;
68 ic2 = 2.0f * v2 - ic2;
69 return {v2, v1, input - k * v1 - v2};
70 }
71 void reset()
72 {
73 ic2 = 0.0f;
74 ic1 = 0.0f;
75 }
76
77 private:
78 float ic1 = 0.0f;
79 float ic2 = 0.0f;
80 float k = 1.0f;
81 float a1 = 0.0f;
82 float a2 = 0.0f;
83 float a3 = 0.0f;
84};
