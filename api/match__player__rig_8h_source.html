58inline TwoBone solveTwoBone(Vec3 root, Vec3 target, float upper, float lower,
59 Vec3 pole)
60{
61 const Vec3 toTarget = target - root;
62 float distance = RenderMath::length(toTarget);
63 const Vec3 direction =
64 distance > 1e-5f ? toTarget * (1.0f / distance) : Vec3{0.0f, 0.0f, -1.0f};
65 const float minimum = std::abs(upper - lower) + 1e-3f;
66 const float maximum = (upper + lower) * 0.9995f;
67 const bool reached = distance >= minimum && distance <= maximum;
68 distance = std::clamp(distance, minimum, maximum);
69 const float along =
70 (upper * upper - lower * lower + distance * distance) / (2.0f * distance);
71 const float height = std::sqrt(std::max(upper * upper - along * along, 0.0f));
72 Vec3 bend = pole - direction * RenderMath::dot(pole, direction);
73 const float bendLength = RenderMath::length(bend);
74 if (bendLength < 1e-4f)
75 {
76 // Pole along the limb: any perpendicular will do.
77 bend = RenderMath::cross(
78 direction, std::abs(direction.z) < 0.9f ? UP : Vec3{1.0f, 0.0f, 0.0f});
79 bend = RenderMath::normalize(bend);
80 }
81 else
82 {
83 bend = bend * (1.0f / bendLength);
84 }
85 return {root + direction * along + bend * height, root + direction * distance,
86 reached};
87}

95inline float rollOntoToes(Vec3& ankle, Vec3 hip, Vec3 footForward,
96 float toeLength, float reach, float maxAngle)
97{
98 if (RenderMath::length(ankle - hip) <= reach) return 0.0f;
99 const Vec3 toe = ankle + footForward * toeLength;
100 const auto at = [&](float angle)
101 {
102 return toe - footForward * (toeLength * std::cos(angle)) +
103 UP * (toeLength * std::sin(angle));
104 };
105 float low = 0.0f;
106 float high = maxAngle;
107 if (RenderMath::length(at(high) - hip) > reach)
108 {
109 ankle = at(high);
110 return high;
111 }
112 for (int step = 0; step < 8; ++step)
113 {
114 const float middle = (low + high) * 0.5f;
115 if (RenderMath::length(at(middle) - hip) > reach)
116 low = middle;
117 else
118 high = middle;
119 }
120 ankle = at(high);
121 return high;
122}

176inline Vec3 stanceSpot(const StrideInput& input, float share)
177{
178 const float front = std::min(input.stanceLength * 0.5f, input.frontReach);
179 return input.rest + input.forward * (front - share * input.stanceLength);

189inline Vec3 stepFoot(FootState& foot, const StrideInput& input)
190{
191 if (!foot.valid)
192 {
193 // Standing: both feet down under the hips. Moving: the gait below puts
194 // the foot down on its exact spot (or swings it from under the body).
195 foot = FootState{};
196 foot.position = foot.planted = foot.liftOff =
197 input.idle ? input.rest : stanceSpot(input, 0.5f);
198 foot.yaw = input.yaw;
199 foot.inStance = input.idle;
200 foot.valid = true;
201 }
202 if (input.idle)
203 {
204 if (foot.stepSeconds > 0.0f)
205 {
206 foot.stepSeconds = std::max(0.0f, foot.stepSeconds - input.deltaSeconds);
207 const float share = 1.0f - foot.stepSeconds / input.stepDuration;
208 foot.position =
209 RenderMath::lerp(foot.liftOff, foot.stepTarget, smoothStep(share)) +
210 UP * (input.liftHeight * 0.5f * std::sin(PI * share));
211 if (foot.stepSeconds <= 0.0f)
212 {
213 foot.position = foot.planted = foot.stepTarget;
214 foot.yaw = input.yaw;
215 foot.inStance = true;
216 }
217 return foot.position;
218 }
219 const bool drifted =
220 flatDistance(foot.planted, input.rest) > input.idleStepMetres;
221 if ((!foot.inStance || drifted) && !input.otherStepping)
222 {
223 // Step back under the hip (a foot caught mid-swing lands there too).
224 foot.liftOff = foot.position;
225 foot.stepTarget = input.rest;
226 foot.stepSeconds = input.stepDuration;
227 foot.inStance = false;
228 return foot.position;
229 }
230 if (foot.inStance) foot.position = foot.planted;
231 return foot.position;
232 }
233
234 if (foot.stepSeconds > 0.0f)
235 {
236 // Started to move during an idle step: carry on as a swing.
237 foot.stepSeconds = 0.0f;
238 foot.inStance = false;
239 foot.liftOff = foot.position;
240 }
241 const float share = std::clamp(input.legPhase / TWO_PI, 0.0f, 1.0f);
242 const bool stance = share < input.duty;
243 if (stance)
244 {
245 const Vec3 ideal = stanceSpot(input, share / std::max(input.duty, 1e-3f));
246 if (!foot.inStance || flatDistance(foot.planted, ideal) > input.maxDrift)
247 {
248 foot.planted = ideal;
249 foot.yaw = input.yaw;
250 }
251 foot.position = foot.planted;
252 }
253 else
254 {
255 if (foot.inStance) foot.liftOff = foot.planted;
256 const float swing =
257 (share - input.duty) / std::max(1.0f - input.duty, 1e-3f);
258 const Vec3 landing = stanceSpot(input, 0.0f);
259 foot.position = RenderMath::lerp(foot.liftOff, landing, smoothStep(swing)) +
260 UP * (input.liftHeight * std::sin(PI * swing));
261 foot.yaw = input.yaw;
262 }
263 foot.inStance = stance;
264 return foot.position;
265}

342inline Action locomotionFor(float speed, const GaitBands& bands = {})
343{
344 if (speed < bands.idle) return Action::IDLE;
345 if (speed < bands.walk) return Action::WALK;
346 if (speed < bands.jog) return Action::JOG;
347 if (speed < bands.run) return Action::RUN;
348 return Action::SPRINT;
349}

356inline Action selectAction(const ActionInput& input,
357 const GaitBands& bands = {})
358{
359 if (input.goalkeeper && input.diving) return Action::KEEPER_DIVE;
360 switch (input.event)
361 {
362 case Event::SLIDE:
363 return Action::SLIDE;
364 case Event::TACKLE:
365 return Action::TACKLE;
366 case Event::HEADER:
367 return Action::HEADER;
368 case Event::SHOT:
369 return Action::SHOT;
370 case Event::CROSS:
371 return Action::CROSS;
372 case Event::PASS:
373 return Action::PASS;
374 case Event::THROW:
375 return Action::THROW_IN;
376 case Event::NONE:
377 break;
378 }
379 if (input.throwIn) return Action::THROW_IN;
380 if (input.goalkeeper && input.holding) return Action::KEEPER_HOLD;
381 if (input.mood == Mood::CELEBRATE) return Action::CELEBRATE;
382 if (input.mood == Mood::DEJECTED) return Action::DEJECTED;
383 if (input.goalkeeper && input.keeperSet && input.speed < bands.jog)
384 return Action::KEEPER_SET;
385 if (std::abs(input.turnRate) >= bands.turnRate && input.speed >= bands.idle &&
386 input.speed < bands.turnSpeed)
387 return Action::TURN;
388 return locomotionFor(input.speed, bands);
389}

392inline float dutyFactor(float speed)
393{
394 // Walking keeps a foot down most of the time; running has flight phases.
395 constexpr float WALK_DUTY = 0.62f;
396 constexpr float JOG_DUTY = 0.4f;
397 constexpr float SPRINT_DUTY = 0.28f;
398 if (speed <= 1.5f) return WALK_DUTY;
399 if (speed <= 4.0f)
400 return WALK_DUTY + (JOG_DUTY - WALK_DUTY) * (speed - 1.5f) / 2.5f;
401 return std::max(SPRINT_DUTY,
402 JOG_DUTY + (SPRINT_DUTY - JOG_DUTY) * (speed - 4.0f) / 3.5f);
403}

430struct KickTiming
431{
432 float contactHold = 0.05f;
433 float followThrough = 0.26f;
434 float recover = 0.26f;
435 float total() const { return contactHold + followThrough + recover; }
436};
437
444inline Vec3 kickAnkle(Event event, float seconds, Vec3 contact, Vec3 direction,
445 Vec3 across, const KickTiming& timing, float& weight)
446{
447 weight = 0.0f;
448 if (seconds < 0.0f || seconds >= timing.total()) return contact;
449 const Swing swing = followThroughFor(event);
450 const Vec3 finish = contact + direction * swing.forward +
451 across * swing.across + UP * swing.up;
452 if (seconds < timing.contactHold)
453 {
454 weight = 1.0f;
455 return contact;
456 }
457 const float through = (seconds - timing.contactHold) / timing.followThrough;
458 if (through < 1.0f)
459 {
460 weight = 1.0f;
461 // Fast off the ball, slowing towards the top of the follow-through.
462 const float eased = 1.0f - (1.0f - through) * (1.0f - through);
463 return RenderMath::lerp(contact, finish, eased);
464 }
465 const float back =
466 (seconds - timing.contactHold - timing.followThrough) / timing.recover;
467 weight = 1.0f - smoothStep(back);
468 return finish;
469}
