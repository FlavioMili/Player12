13// A retaining contact changes velocity, not position. A ball whose centre is
14// slightly over a line can still be legally in play. Preserve the legacy inset
15// only for Aardvark; Binturong must not snap such a touch back onto the pitch.
16constexpr PitchPoint retainingTouchPosition(const MatchPitch& pitch,
17 PossessionId version,
18 PitchPoint contact)
19{
20 if (version == PossessionId::BINTURONG)
21 {
22 return contact;
23 }
24 return PitchPoint{
25 std::clamp(contact.x, MatchTuning::Pitch::BALL_INSIDE_END_METRES,
26 pitch.length - MatchTuning::Pitch::BALL_INSIDE_END_METRES),
27 std::clamp(contact.y, MatchTuning::Pitch::BALL_INSIDE_SIDE_METRES,
28 pitch.width - MatchTuning::Pitch::BALL_INSIDE_SIDE_METRES)};
29}
30// Initial Binturong policy values, in metres and simulated seconds. These
31// describe current control, not claim expiry or permission to contest a ball.
32inline constexpr float RETAIN_GAP_METRES = 3.0F;
33inline constexpr float CLEAR_LOSS_GAP_METRES = 4.5F;
34inline constexpr float PREDICT_SECONDS = 0.35F;
35inline constexpr float HEIGHT_TOLERANCE_METRES = 0.15F;
36inline constexpr float CLEAR_LOSS_HEIGHT_METRES = 0.75F;
37inline constexpr float SEPARATION_GRACE_SECONDS = 0.20F;
38
39struct Evidence
40{
41 bool playerAvailable = false;

46 float feetHeightMetres = 0.0F;

49};

51struct Decision

58// A wider retention envelope cannot establish control: collection must first
59// meet foot reach, height and relative-speed evidence. Predicting relative
60// separation accounts for both actors' speed/direction without moving the ball.
61constexpr Decision evaluate(bool collected, float separationSeconds,
62 const Evidence& evidence, float dt)
63{
64 const bool collectable =
65 evidence.playerAvailable &&
66 evidence.gapMetres <= evidence.collectionReachMetres &&
67 evidence.relativeSpeed <= evidence.collectionRelativeSpeed &&
68 evidence.heightMetres <= evidence.feetHeightMetres;
69 if (!collected)
70 {
71 return {.collected = collectable,
72 .separationSeconds = 0.0F,
73 .newlyCollected = collectable};
74 }
75
76 const bool clearlyLost =
77 !evidence.playerAvailable || evidence.gapMetres > CLEAR_LOSS_GAP_METRES ||
78 evidence.predictedGapMetres > CLEAR_LOSS_GAP_METRES ||
79 evidence.heightMetres >
80 evidence.feetHeightMetres + CLEAR_LOSS_HEIGHT_METRES;
81 if (clearlyLost)
82 {
83 return {};
84 }
85
86 const bool retained = evidence.gapMetres <= RETAIN_GAP_METRES &&
87 evidence.predictedGapMetres <= RETAIN_GAP_METRES &&
88 evidence.heightMetres <=
89 evidence.feetHeightMetres + HEIGHT_TOLERANCE_METRES;
90 if (retained)
91 {
92 return {
93 .collected = true, .separationSeconds = 0.0F, .newlyCollected = false};
94 }
95
96 const float separated =
97 std::max(0.0F, separationSeconds) + std::max(0.0F, dt);
98 return separated + 1e-6F < SEPARATION_GRACE_SECONDS
99 ? Decision{.collected = true,
100 .separationSeconds = separated,
101 .newlyCollected = false}
102 : Decision{};
103}
