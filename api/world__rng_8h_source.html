144template <typename Engine>
145concept FullRangeEngine =
146 Engine::min() == 0 &&
147 (Engine::max() == std::numeric_limits<std::uint32_t>::max() ||
148 Engine::max() == std::numeric_limits<std::uint64_t>::max());
149
151template <FullRangeEngine Engine>
152std::uint32_t bits32(Engine& engine)
153{
154 if constexpr (Engine::max() == std::numeric_limits<std::uint32_t>::max())
155 return static_cast<std::uint32_t>(engine());
156 else
157 return static_cast<std::uint32_t>(static_cast<std::uint64_t>(engine()) >>
158 32U);
159}
160
162template <FullRangeEngine Engine>
163std::uint64_t bits64(Engine& engine)
164{
165 if constexpr (Engine::max() == std::numeric_limits<std::uint32_t>::max())
166 {
167 const auto high = static_cast<std::uint64_t>(engine());
168 return (high << 32U) | static_cast<std::uint64_t>(engine());
169 }
170 else
171 {
172 return static_cast<std::uint64_t>(engine());
173 }
174}

205template <FullRangeEngine Engine>
206std::uint64_t below(Engine& engine, std::uint64_t count)
207{
208 if (count <= 1) return 0;
209 // Rejection sampling: values under the threshold would favour the low
210 // residues.
211 const std::uint64_t threshold = (0U - count) % count;
212 std::uint64_t value = bits64(engine);
213 while (value < threshold) value = bits64(engine);
214 return value % count;
215}
216
218template <FullRangeEngine Engine>
219int uniformInt(Engine& engine, int lo, int hi)
220{
221 if (hi <= lo) return lo;
222 const auto count =
223 static_cast<std::uint64_t>(static_cast<std::int64_t>(hi) - lo) + 1U;
224 return static_cast<int>(static_cast<std::int64_t>(lo) +
225 static_cast<std::int64_t>(below(engine, count)));
226}
227
229template <FullRangeEngine Engine>
230double normal(Engine& engine, double mean, double stddev)
231{
232 // 1 - [0, 1) is in (0, 1], so the logarithm is always finite.
233 const double u1 = 1.0 - unitDouble(bits64(engine));
234 const double u2 = unitDouble(bits64(engine));
235 return affine(mean, stddev, standardNormal(u1, u2));
236}
237
239template <FullRangeEngine Engine>
240int poisson(Engine& engine, double mean)
241{
242 // A sum of independent Poisson counts is a Poisson count of the summed
243 // means, so large means are split into parts inversion handles well.
244 constexpr double PART = 64.0;
245 int count = 0;
246 while (mean > PART)
247 {
248 count += poissonFromUnit(unitDouble(bits64(engine)), PART);
249 mean -= PART;
250 }
251 return count + poissonFromUnit(unitDouble(bits64(engine)), mean);
252}
