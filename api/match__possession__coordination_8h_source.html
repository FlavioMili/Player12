32// Current motion is evidence, not a promise that the player can reach a point.
33// Positive forward speed into space and running with the delivery improve it.
34inline float quality(Vector2F velocity, Vector2F ballVelocity,
35 bool attacksRight, float opponentSpaceMetres)
36{
37 const float speed = std::hypot(velocity.x, velocity.y);
38 const float ballSpeed = std::hypot(ballVelocity.x, ballVelocity.y);
39 const float forward =
40 std::clamp((attacksRight ? velocity.x : -velocity.x) / 7.0F, 0.0F, 1.0F);
41 const float alignment = speed > 1e-5F && ballSpeed > 1e-5F
42 ? std::clamp((velocity.x * ballVelocity.x +
43 velocity.y * ballVelocity.y) /
44 (speed * ballSpeed),
45 0.0F, 1.0F)
46 : 0.0F;
47 const float space = std::clamp(opponentSpaceMetres / 8.0F, 0.0F, 1.0F);
48 return (0.75F * forward * (0.5F + 0.5F * space)) +
49 (0.25F * std::clamp(speed / 7.0F, 0.0F, 1.0F) * alignment);
50}

52inline float cost(const Candidate& candidate)
53{
54 return std::floor((candidate.arrivalSeconds -
55 COLLECTION_ADVANTAGE_SECONDS *
56 std::clamp(candidate.collectionQuality, 0.0F, 1.0F)) /
57 TIE_SECONDS);
58}
59inline bool better(const Candidate& a, const Candidate& b)
60{
61 const float ac = cost(a), bc = cost(b);
62 if (ac != bc)
63 {
64 return ac < bc;
65 }
66 if (a.intendedReceiver != b.intendedReceiver)
67 {
68 return a.intendedReceiver;
69 }
70 return a.id < b.id;
71}

73// The input is chronological. Coordinate only contiguous playable contacts of
74// one side: never move a contact past an opponent, body collision or shot
75// block. If the preferred player declines/misses, the remaining contacts still
76// run.
77template <typename Contact, std::size_t N, typename Eligible, typename SameTeam,
78 typename MakeCandidate>
79void prioritise(std::array<Contact, N>& contacts, std::size_t count,
80 Eligible eligible, SameTeam sameTeam, MakeCandidate candidate)
81{
82 for (std::size_t first = 0; first < count;)
83 {
84 if (!eligible(contacts[first]))
85 {
86 ++first;
87 continue;
88 }
89 const float deadline = candidate(contacts[first]).arrivalSeconds +
90 COLLECTION_ADVANTAGE_SECONDS;
91 std::size_t end = first + 1;
92 while (end < count && eligible(contacts[end]) &&
93 sameTeam(contacts[first], contacts[end]) &&
94 candidate(contacts[end]).arrivalSeconds <= deadline)
95 {
96 ++end;
97 }
98 std::stable_sort(contacts.begin() + first, contacts.begin() + end,
99 [&](const Contact& a, const Contact& b)
100 { return better(candidate(a), candidate(b)); });
101 first = end;
102 }
103}
