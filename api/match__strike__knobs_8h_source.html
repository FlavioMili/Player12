11// Strike knobs: every value of a strike's contact profile (Binturong body,
12// Notes and plans/match_engine_contact_plan.md §3 and §5), per kind of
13// strike, as runtime values each engine holds. They default to the
14// MatchTuning constants, so a match nobody sets them on plays exactly as
15// before; drills and the lab change them. A knob's full name is "<kind>:
16// <name>", e.g. "Ground pass: Angle fail base".
17//
18// To add a knob: an id in StrikeKnob (before COUNT), a row in STRIKE_KNOBS
19// in the same place, its default in strikeProfileDefault (and every
20// kind's profile in MatchTuning::Strike), then read it in
21// strikeProfileBinturong.

533constexpr float strikeKnobDefault(StrikeKind kind, StrikeKnob knob)
534{
535 using S = MatchTuning::Strike;

537 {
538 case StrikeKind::GROUND_PASS:
539 return strikeProfileDefault<S::GroundPass>(knob);
540 case StrikeKind::LOFTED_PASS:
541 return strikeProfileDefault<S::Lofted>(knob);
542 case StrikeKind::CROSS:
543 return strikeProfileDefault<S::Cross>(knob);
544 case StrikeKind::SHOT:
545 return strikeProfileDefault<S::Shot>(knob);
546 case StrikeKind::HEADER:
547 return strikeProfileDefault<S::Header>(knob);
548 case StrikeKind::CLEARANCE:
549 return strikeProfileDefault<S::Clearance>(knob);
550 case StrikeKind::THROW:
551 return strikeProfileDefault<S::Throw>(knob);
552 case StrikeKind::DRIVEN_PASS:
553 return strikeProfileDefault<S::DrivenPass>(knob);
554 case StrikeKind::CHIP:
555 return strikeProfileDefault<S::Chip>(knob);
556 case StrikeKind::PLACED_SHOT:
557 return strikeProfileDefault<S::PlacedShot>(knob);
558 case StrikeKind::CURLED_SHOT:
559 return strikeProfileDefault<S::CurledShot>(knob);
560 case StrikeKind::CHIP_SHOT:

576struct StrikeKnobs
577{
578 std::array<std::array<float, STRIKE_KNOB_COUNT>, STRIKE_KIND_COUNT> values =
579 []
580 {
581 std::array<std::array<float, STRIKE_KNOB_COUNT>, STRIKE_KIND_COUNT>
582 defaults{};
583 for (std::size_t kind = 0; kind < STRIKE_KIND_COUNT; ++kind)
584 {
585 for (std::size_t knob = 0; knob < STRIKE_KNOB_COUNT; ++knob)
586 {
587 defaults[kind][knob] = strikeKnobDefault(static_cast<StrikeKind>(kind),
588 static_cast<StrikeKnob>(knob));
589 }
590 }
591 return defaults;
592 }();
593
594 float at(StrikeKind kind, StrikeKnob knob) const
595 {
596 return values[static_cast<std::size_t>(kind)]
597 [static_cast<std::size_t>(knob)];
598 }
599 float& at(StrikeKind kind, StrikeKnob knob)
600 {
601 return values[static_cast<std::size_t>(kind)]
602 [static_cast<std::size_t>(knob)];
603 }
606 float* find(std::string_view fullName)
607 {
608 for (std::size_t kind = 0; kind < STRIKE_KIND_COUNT; ++kind)
609 {
610 for (std::size_t knob = 0; knob < STRIKE_KNOB_COUNT; ++knob)
611 {
612 if (strikeKnobName(static_cast<StrikeKind>(kind),
613 static_cast<StrikeKnob>(knob)) == fullName)
614 {
615 return &values[kind][knob];
616 }
617 }
618 }
619 return nullptr;
620 }
621};
