29// A conservative radial arrival estimate. Moving away costs time before the
30// player can make progress toward this point; acceleration is capped at top
31// speed.
32inline float travelBudget(float towardSpeed, float topSpeed, float acceleration,
33 float seconds)
34{
35 const float top = std::max(0.0F, topSpeed);
36 const float initial = std::clamp(towardSpeed, -top, top);
37 const float time = std::max(0.0F, seconds);
38 const float a = std::max(0.0F, acceleration);
39 if (a <= 1e-6F)
40 {
41 return std::max(0.0F, initial * time);
42 }
43 const float accelerating = std::min(time, (top - initial) / a);
44 return std::max(0.0F, (initial * accelerating) +
45 (0.5F * a * accelerating * accelerating) +
46 (top * (time - accelerating)));
47}

49inline bool canRecoverAt(float distanceMetres, float towardSpeed,
50 float topSpeed, float acceleration, float seconds,
51 float contactReadyIn, float reachMetres,
52 float heightMetres, float feetHeight, bool onPitch)
53{
54 return onPitch && seconds + 1e-6F >= std::max(0.0F, contactReadyIn) &&
55 heightMetres <= feetHeight &&
56 distanceMetres <=
57 std::max(0.0F, reachMetres) +
58 travelBudget(towardSpeed, topSpeed, acceleration, seconds);
59}

61constexpr Decision evaluate(bool available, bool controlled, bool recoverable,
62 float age, float unrecoverable, float dt)
63{
64 if (!available)
65 {
66 return {.uncontrolledSeconds = 0.0F,
67 .unrecoverableSeconds = 0.0F,
68 .expiry = Expiry::UNAVAILABLE};
69 }
70 if (controlled)
71 {
72 return {};
73 }
74 const float elapsed = std::max(0.0F, dt);
75 const float nextAge = std::max(0.0F, age) + elapsed;
76 const float nextUnrecoverable =
77 recoverable ? 0.0F : std::max(0.0F, unrecoverable) + elapsed;
78 const auto reason = nextAge + 1e-6F >= MAX_UNCONTROLLED_SECONDS
79 ? Expiry::AGE_LIMIT
80 : nextUnrecoverable + 1e-6F >= UNRECOVERABLE_GRACE_SECONDS
81 ? Expiry::UNRECOVERABLE
82 : Expiry::NONE;
83 return {.uncontrolledSeconds = nextAge,
84 .unrecoverableSeconds = nextUnrecoverable,
85 .expiry = reason};
86}
