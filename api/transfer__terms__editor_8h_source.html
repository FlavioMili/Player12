50template <typename T, typename V>
51int optionIndex(std::span<const T> options, V value)
52{
53 std::size_t best = 0;
54 for (std::size_t index = 1; index < options.size(); ++index)
55 {
56 if (std::abs(static_cast<int>(options[index]) - static_cast<int>(value)) <
57 std::abs(static_cast<int>(options[best]) - static_cast<int>(value)))
58 best = index;
59 }
60 return static_cast<int>(best);
61}

63template <typename T, std::size_t N, typename V>
64int optionIndex(const std::array<T, N>& options, V value)
65{
66 return optionIndex(std::span<const T>(options), value);
67}
