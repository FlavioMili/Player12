54struct DirectionMix
55{
56 std::size_t below = 0;
57 float onward = 0.0F;
58};
59inline DirectionMix directionMix(float angle)
60{
61 constexpr float STEP = 0.25F * 3.14159265F;
62 const float steps = std::clamp(angle / STEP, 0.0F, 4.0F);
63 const std::size_t below =
64 std::min<std::size_t>(static_cast<std::size_t>(steps), 3);
65 return {.below = below, .onward = steps - static_cast<float>(below)};
66}
68template <typename Table>
69inline float directionValue(const Table& table, DirectionMix mix)
70{
71 return static_cast<float>(table[mix.below]) +
72 ((static_cast<float>(table[mix.below + 1]) -
73 static_cast<float>(table[mix.below])) *
74 mix.onward);
75}

114inline PitchPoint keepOutOfGoalFrames(const MatchPitch& pitch, PitchPoint from,
115 PitchPoint to)
116{
117 using P = MatchTuning::Pitch;
118 constexpr float R = MatchTuning::Body::RADIUS_METRES;
119 const float W = pitch.goal.halfWidth + (0.5F * pitch.goal.post);
120 const float D = pitch.goal.netDepth;
121 // Local metres at the nearer goal: u behind its line, v across from its
122 // centre.
123 const bool right = to.x > pitch.atX(P::CENTRE);
124 const float lineX = right ? pitch.length : 0.0F;
125 const float behind = right ? 1.0F : -1.0F;
126 const float u0 = (from.x - lineX) * behind;
127 const float v0 = from.y - pitch.atY(P::CENTRE);
128 float u = (to.x - lineX) * behind;
129 float v = to.y - pitch.atY(P::CENTRE);
130 // Nowhere near this frame, before or after: nothing to meet.
131 if (std::max(u0, u) < -R || std::min(u0, u) > D + R ||
132 std::min(v0, v) > W + R || std::max(v0, v) < -(W + R))
133 {
134 return to;
135 }
136 bool movedU = false;
137 bool movedV = false;
138 // The side nets, post to back net.
139 for (const float side : {-W, W})
140 {
141 if (u < 0.0F || u > D)
142 {
143 continue;
144 }
145 const float keep = v0 > side || (v0 == side && side > 0.0F) ? 1.0F : -1.0F;
146 if ((v - side) * keep < R)
147 {
148 v = side + keep * R;
149 movedV = true;
150 }
151 }
152 // The back net.
153 if (std::abs(v) <= W)
154 {
155 const float keep = u0 >= D ? 1.0F : -1.0F;
156 if ((u - D) * keep < R)
157 {
158 u = D + keep * R;
159 movedU = true;
160 }
161 }
162 // The posts and the back corners: round them.
163 for (const float cornerU : {0.0F, D})
164 {
165 for (const float cornerV : {-W, W})
166 {
167 const float du = u - cornerU;
168 const float dv = v - cornerV;
169 const float gap = std::sqrt((du * du) + (dv * dv));
170 if (gap >= R)
171 {
172 continue;
173 }
174 const float fromU = u0 - cornerU;
175 const float fromV = v0 - cornerV;
176 const float fromGap = std::sqrt((fromU * fromU) + (fromV * fromV));
177 // On it: back out the way he came (or out of the goal).
178 const float awayU = gap > 1e-4F ? du / gap
179 : fromGap > 1e-4F ? fromU / fromGap
180 : -1.0F;
181 const float awayV = gap > 1e-4F ? dv / gap
182 : fromGap > 1e-4F ? fromV / fromGap
183 : 0.0F;
184 u = cornerU + awayU * R;
185 v = cornerV + awayV * R;
186 movedU = true;
187 movedV = true;
188 }
189 }
190 return PitchPoint{movedU ? lineX + (u * behind) : to.x,
191 movedV ? pitch.atY(P::CENTRE) + v : to.y};
192}

201inline bool keepOffWall(bool alongX, float plane, float low, float high,
202 Vector2F from, Vector2F& to)
203{
204 constexpr float R = MatchTuning::Body::RADIUS_METRES;
205 const float across0 = alongX ? from.y : from.x;
206 float& across1 = alongX ? to.y : to.x;
207 const float along0 = alongX ? from.x : from.y;
208 const float along1 = alongX ? to.x : to.y;
209 const float keep = across0 < plane ? -1.0F : 1.0F;
210 const float face = plane + (keep * R);
211 if ((across1 - face) * keep >= 0.0F)
212 {
213 return false;
214 }
215 // Where along the wall his path reaches its face (where he starts, if he
216 // is already over it).
217 const float span = across1 - across0;
218 const float t = (across0 - face) * keep > 0.0F && std::abs(span) > 1e-6F
219 ? (face - across0) / span
220 : 0.0F;
221 const float along = along0 + ((along1 - along0) * t);
222 if (along < low || along > high)
223 {
224 return false;
225 }
226 across1 = face;
227 return true;
228}

232inline bool keepOffWallEnd(Vector2F end, Vector2F from, Vector2F& to)
233{
234 constexpr float R = MatchTuning::Body::RADIUS_METRES;
235 const float dx = to.x - end.x;
236 const float dy = to.y - end.y;
237 const float gap = std::sqrt((dx * dx) + (dy * dy));
238 if (gap >= R)
239 {
240 return false;
241 }
242 const float fromX = from.x - end.x;
243 const float fromY = from.y - end.y;
244 const float fromGap = std::sqrt((fromX * fromX) + (fromY * fromY));
245 // On it: back out the way he came.
246 const float awayX = gap > 1e-4F ? dx / gap
247 : fromGap > 1e-4F ? fromX / fromGap
248 : 0.0F;
249 const float awayY = gap > 1e-4F ? dy / gap
250 : fromGap > 1e-4F ? fromY / fromGap
251 : -1.0F;
252 to = {.x = end.x + (awayX * R), .y = end.y + (awayY * R)};
253 return true;
254}

264inline PitchPoint keepOffGround(const MatchPitch& pitch, PitchPoint from,
265 PitchPoint to)
266{
267 using P = MatchTuning::Pitch;
268 const MatchGround::Layout& G = pitch.ground;
269 constexpr float R = MatchTuning::Body::RADIUS_METRES;
270 // A walled pitch: the walls, and nothing beyond them (no furniture). A
271 // body stays on the side of the walls it is on: the players inside, a
272 // coach at his station outside (a radius off them).
273 if (pitch.edge == PitchEdge::WALLS)
274 {
275 if (from.x >= pitch.bodyMinX && from.x <= pitch.bodyMaxX &&
276 from.y >= pitch.bodyMinY && from.y <= pitch.bodyMaxY)
277 {
278 return PitchPoint{std::clamp(to.x, pitch.bodyMinX, pitch.bodyMaxX),
279 std::clamp(to.y, pitch.bodyMinY, pitch.bodyMaxY)};
280 }
281 PitchPoint out = to;
282 if (from.y > pitch.bodyMaxY)
283 {
284 out.y = std::max(out.y, pitch.width + R);
285 }
286 else if (from.y < pitch.bodyMinY)
287 {
288 out.y = std::min(out.y, -R);
289 }
290 else if (from.x > pitch.bodyMaxX)
291 {
292 out.x = std::max(out.x, pitch.length + R);
293 }
294 else
295 {
296 out.x = std::min(out.x, -R);
297 }
298 return out;
299 }
300 // Inside the boards, before and after: nothing to meet.
301 const auto insideBoards = [&pitch](PitchPoint at)
302 {
303 return at.x >= pitch.bodyMinX && at.x <= pitch.bodyMaxX &&
304 at.y >= pitch.bodyMinY && at.y <= pitch.bodyMaxY;
305 };
306 if (insideBoards(from) && insideBoards(to))
307 {
308 return to;
309 }
310 const Vector2F start{.x = from.x, .y = from.y};
311 const Vector2F wanted{.x = to.x, .y = to.y};
312 // The stands' front walls (and the tunnel's far end) all round.
313 Vector2F end{
314 .x = std::clamp(wanted.x, -P::STAND_END_METRES + R,
315 pitch.length + P::STAND_END_METRES - R),
316 .y = std::clamp(wanted.y, -P::STAND_SIDE_METRES + R, G.tunnel.y1 - R)};
317 bool moved = end.x != wanted.x || end.y != wanted.y;
318 // The boards: whole along the top and the ends, open along the bottom.
319 moved |= keepOffWall(true, -P::BOARDS_SIDE_METRES, -P::BOARDS_END_METRES,
320 pitch.length + P::BOARDS_END_METRES, start, end);
321 for (const float plane :
322 {-P::BOARDS_END_METRES, pitch.length + P::BOARDS_END_METRES})
323 {
324 moved |= keepOffWall(false, plane, -P::BOARDS_SIDE_METRES, G.boardsY, start,
325 end);
326 }
327 for (const MatchGround::Span& span : G.bottomBoards)
328 {
329 moved |= keepOffWall(true, G.boardsY, span.from, span.to, start, end);
330 moved |= keepOffWallEnd({.x = span.from, .y = G.boardsY}, start, end);
331 moved |= keepOffWallEnd({.x = span.to, .y = G.boardsY}, start, end);
332 }
333 // The dugouts' fronts (their backs are the stand wall).
334 for (const MatchGround::Rect& dugout : G.dugouts)
335 {
336 moved |= keepOffWall(true, dugout.y0, dugout.x0, dugout.x1, start, end);
337 moved |= keepOffWallEnd({.x = dugout.x0, .y = dugout.y0}, start, end);
338 moved |= keepOffWallEnd({.x = dugout.x1, .y = dugout.y0}, start, end);
339 }
340 // The bottom stand's front wall, open at the tunnel mouth, and the
341 // tunnel's sides.
342 for (const MatchGround::Span& span : G.bottomStandWall)
343 {
344 moved |= keepOffWall(true, G.standY, span.from, span.to, start, end);
345 }
346 for (const float side : {G.tunnel.x0, G.tunnel.x1})
347 {
348 moved |= keepOffWall(false, side, G.tunnel.y0, G.tunnel.y1, start, end);
349 moved |= keepOffWallEnd({.x = side, .y = G.tunnel.y0}, start, end);
350 }
351 if (!moved)
352 {
353 return to;
354 }
355 return PitchPoint{end.x != wanted.x ? end.x : to.x,
356 end.y != wanted.y ? end.y : to.y};
357}

386inline Vector2F routeViaTunnel(const MatchPitch& pitch, Vector2F at,
387 Vector2F target)
388{
389 const MatchGround::Layout& G = pitch.ground;
390 constexpr float MARGIN = 0.6F;
391 const auto inLane = [&G](Vector2F p)
392 { return p.x > G.tunnel.x0 && p.x < G.tunnel.x1 && p.y > G.boardsY - 0.5F; };
393 const auto inTunnel = [&G](Vector2F p) { return p.y > G.standY - 0.2F; };
394 const float laneX =
395 std::clamp(at.x, G.tunnel.x0 + MARGIN, G.tunnel.x1 - MARGIN);
396 // Coming out: up the tunnel, then through the gap.
397 if (inTunnel(at) && !inTunnel(target))
398 {
399 return {.x = laneX, .y = G.standY - 1.0F};
400 }
401 if (inLane(at) && target.y < G.boardsY - 0.5F)
402 {
403 return {.x = laneX, .y = G.boardsY - 1.5F};
404 }
405 // Going in: to the gap first, unless already lined up with it.
406 if (inTunnel(target) && !inLane(at))
407 {
408 const float x =
409 std::clamp(target.x, G.tunnel.x0 + MARGIN, G.tunnel.x1 - MARGIN);
410 const bool lined = std::abs(at.x - x) < 0.5F && at.y > G.boardsY - 3.0F;
411 if (!lined)
412 {
413 return {.x = x, .y = G.boardsY - 1.5F};
414 }
415 }
416 return target;
417}

423inline Vector2F nearestExit(const MatchPitch& pitch, Vector2F at)
424{
425 const MatchGround::Layout& G = pitch.ground;
426 const float out = MatchTuning::Body::RADIUS_METRES + 0.3F;
427 const std::array<std::pair<float, Vector2F>, 4> exits{{
428 {at.y, {.x = at.x, .y = -out}},
429 {pitch.width - at.y, {.x = at.x, .y = pitch.width + out}},
430 {at.x, {.x = -out, .y = at.y}},
431 {pitch.length - at.x, {.x = pitch.length + out, .y = at.y}},
432 }};
433 Vector2F exit =
434 std::ranges::min(exits, {}, &std::pair<float, Vector2F>::first).second;
435 const float CLEAR_OF_POST = pitch.goal.halfWidth + 1.5F;
436 if ((exit.x < 0.0F || exit.x > pitch.length) &&
437 std::abs(exit.y - (0.5F * pitch.width)) < CLEAR_OF_POST)
438 {
439 exit.y = 0.5F * pitch.width +
440 (exit.y < 0.5F * pitch.width ? -CLEAR_OF_POST : CLEAR_OF_POST);
441 }
442 return exit;
443}

467inline float fastAtan2(float y, float x)
468{
469 const float absX = std::abs(x);
470 const float absY = std::abs(y);
471 if (absX < 1e-12F && absY < 1e-12F)
472 {
473 return 0.0F;
474 }
475 const float ratio = std::min(absX, absY) / std::max(absX, absY);
476 const float squared = ratio * ratio;
477 float angle =
478 (((-0.0464964749F * squared + 0.15931422F) * squared - 0.327622764F) *
479 squared * ratio) +
480 ratio;
481 if (absY > absX)
482 {
483 angle = 1.57079637F - angle;
484 }
485 if (x < 0.0F)
486 {
487 angle = 3.14159274F - angle;
488 }
489 return y < 0.0F ? -angle : angle;
490}

517inline float uniform01(std::mt19937& engine)
518{
519 // Portable mapping of the 32-bit output to [0, 1) (24 bits of mantissa).
520 return static_cast<float>(engine() >> 8U) * 0x1p-24F;

524inline float gaussian(std::mt19937& engine)
525{
526 const float first = std::max(uniform01(engine), 1e-7F);
527 const float second = uniform01(engine);
528 return std::sqrt(-2.0F * std::log(first)) * std::cos(TWO_PI * second);
529}

600template <typename Array, typename Less>
601inline void insertionSort(Array& values, std::size_t count, Less less)
602{
603 for (std::size_t index = 1; index < count; ++index)
604 {
605 auto value = values[index];
606 std::size_t slot = index;
607 while (slot > 0 && less(value, values[slot - 1]))
608 {
609 values[slot] = values[slot - 1];
610 --slot;
611 }
612 values[slot] = value;
613 }
614}

616inline std::uint64_t splitMix64(std::uint64_t value)
617{
618 value += 0x9e3779b97f4a7c15ULL;
619 value = (value ^ (value >> 30U)) * 0xbf58476d1ce4e5b9ULL;
620 value = (value ^ (value >> 27U)) * 0x94d049bb133111ebULL;
621 return value ^ (value >> 31U);
622}
