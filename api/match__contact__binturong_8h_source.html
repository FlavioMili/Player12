66inline float powerByAngleShare(float degrees, float square, float behind)
67{
68 const float off = std::clamp(degrees, 0.0F, 180.0F);
69 if (off <= 90.0F)
70 {
71 const float into = off / 90.0F;
72 return 1.0F - ((1.0F - square) * into * into);
73 }
74 return square + ((behind - square) * (off - 90.0F) / 90.0F);
75}

80inline float powerAskedValue(float speed, float capacity, float comfortShare)
81{
82 if (capacity <= 0.0F)
83 {
84 return 0.0F;
85 }
86 const float share = speed / capacity;
87 if (share <= comfortShare)
88 {
89 return 0.0F;
90 }
91 const float into = std::min(
92 (share - comfortShare) / std::max(1.0F - comfortShare, 1.0e-3F), 1.0F);
93 return into * into;
94}

99inline bool inTouchDeadZone(Vector2F facing, Vector2F toBall, float gap,
100 float comfortGap, float angleDegrees)
101{
102 if (gap <= comfortGap)
103 {
104 return false;
105 }
106 const float cosine = (facing.x * toBall.x) + (facing.y * toBall.y);
107 return cosine < std::cos(angleDegrees * std::numbers::pi_v<float> / 180.0F);
108}
