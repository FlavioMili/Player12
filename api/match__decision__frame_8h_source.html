81[[nodiscard]] inline float misread(float chance, float logOdds)
82{
83 if (logOdds == 0.0F)
84 {
85 return chance;
86 }
87 constexpr float EDGE = 1.0e-4F;
88 const float clamped = std::clamp(chance, EDGE, 1.0F - EDGE);
89 const float odds = std::log(clamped / (1.0F - clamped)) + logOdds;
90 return 1.0F / (1.0F + std::exp(-odds));
91}
