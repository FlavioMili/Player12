189struct ScoreBreakdown
190{
191 float total = 0.0F;
192 std::vector<ScoreTerm> terms;
193
194 void add(const char* name, TermSource source, float value)
195 {
196 terms.push_back({.name = name, .source = source, .value = value});
197 }
198 [[nodiscard]] float residual() const
199 {
200 float sum = 0.0F;
201 for (const ScoreTerm& term : terms)
202 {
203 sum += term.value;
204 }
205 return total - sum;
206 }
207};
